📋 View Ready-to-Paste Google Apps Script (Auto-Creates Folders in Google Drive) 
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var root = getOrCreateFolder_(DriveApp.getRootFolder(), "EventHub_GameOn_Symposium_2027_Uploads");
  var target = getOrCreateFolder_(root, data.subfolder || "01_Abstracts");
  var bytes = Utilities.base64Decode(data.base64Data);
  var blob = Utilities.newBlob(bytes, data.mimeType || "application/pdf", data.fileName);
  var file = target.createFile(blob);
  return ContentService.createTextOutput(JSON.stringify({ ok: true, url: file.getUrl() }))
    .setMimeType(ContentService.MimeType.JSON);
}
function getOrCreateFolder_(parent, name) {
  var iter = parent.getFoldersByName(name);
  return iter.hasNext() ? iter.next() : parent.createFolder(name);
}
